class MyGATConv(MessagePassing):
"""GAT: alpha_ij = softmax over j of LeakyReLU(a . [W h_i || W h_j])."""
def __init__(self, in_channels, out_channels, heads=8, concat=True,
negative_slope=0.2):
# node_dim=0: propagate carries [N, heads, channels], so the node axis is 0,
# not the default -2 (which would be the head axis here).
super().__init__(aggr="add", node_dim=0)
self.in_channels = in_channels
self.out_channels = out_channels
# Concatenated heads must together be out_channels wide, since the next layer
# is built with in_channels = out_channels.
if concat and out_channels % heads != 0:
heads = 1
self.heads = heads
self.concat = concat
self.head_channels = out_channels // heads if concat else out_channels
self.negative_slope = negative_slope
self.lin = nn.Linear(in_channels, heads * self.head_channels, bias=False)
# a = [a_dst || a_src], split so the edge score is a sum of two node terms.
self.att_src = nn.Parameter(torch.empty(1, heads, self.head_channels))
self.att_dst = nn.Parameter(torch.empty(1, heads, self.head_channels))
self.bias = nn.Parameter(torch.empty(out_channels))
self.reset_parameters()
def reset_parameters(self):
super().reset_parameters()
self.lin.reset_parameters()
nn.init.xavier_uniform_(self.att_src)
nn.init.xavier_uniform_(self.att_dst)
nn.init.zeros_(self.bias)
def forward(self, x, edge_index):
n = x.size(0)
# The softmax denominator runs over N(i) u {i}, as in the GAT paper.
edge_index, _ = remove_self_loops(edge_index)
edge_index, _ = add_self_loops(edge_index, num_nodes=n)
h = self.lin(x).view(n, self.heads, self.head_channels)
out = self.propagate(edge_index, h=h)
# [N, heads, head_channels]
out = out.flatten(1) if self.concat else out.mean(dim=1)
return out + self.bias
def message(self, h_j, h_i, index, size_i):
# a^T [W h_i || W h_j] = a_dst . W h_i + a_src . W h_j.
e = (h_i * self.att_dst).sum(-1) + (h_j * self.att_src).sum(-1)
# [E, heads]
alpha = softmax(F.leaky_relu(e, self.negative_slope), index, num_nodes=size_i)
return h_j * alpha.unsqueeze(-1)
node dim=0 is required because propagate carries an [N, heads, C] tensor; the default −2 would
index the head axis and scatter silently into the wrong rows. The attention vector is stored as
its two halves, so a⊤[Whi∥Whj] becomes adst ·Whi + asrc ·Whj and the concatenation is never
materialised over E edges. Self-loops are added before propagating so the softmax denominator
runs over N(i) ∪{i}, and aggr="add" is correct because αij already sums to one per target.
On the concatenate-or-average instruction: a layer cannot tell from inside whether it is the output
layer, and the two differ in per-head width, not just in the reduction — concatenation needs C/K
features per head, averaging needs C. Since out channels is a contract fixed by the next layer,
I made the choice a constructor default: K = 8 heads of width out channels/K, concatenated.
concat=False gives the averaging variant with full-width heads. The harness constructs every
layer as conv cls(in, out), so every layer here concatenates. With weights copied across, the
layer matches PyG’s GATConv to 0 in floating point.
